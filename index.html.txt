        function getEffectiveCooldown(wIndex) {
            let wData = weapons[wIndex];
            let lvl = weaponSpeedLevels[wIndex] || 0;
            if (wData.isUltimate) {
                let startCool = 60000;
                let minCool = 5000;
                let current = startCool - (lvl / 20) * (startCool - minCool);
                return Math.round(current);
            } else {
                let baseCool = wData.cooldown;
                let multiplier = Math.max(0.1, 1 - (lvl * 0.045));
                return Math.round(baseCool * multiplier);
            }
        }

        function getEffectiveDamage(wIndex) {
            let wData = weapons[wIndex];
            let lvl = weaponDamageLevels[wIndex] || 0;
            let multiplier = 1 + (lvl / 20);
            return Math.round(wData.damage * multiplier);
        }

        function buildDetailedSpiderModel() {
            const rootGroup = new THREE.Group();
            const spider = new THREE.Group();

            abdomen.position.set(0, 1.3, 1.4);
            abdomen.scale.set(1, 0.95, 1.5);
            spider.add(abdomen);

            for(let dz = 0.8; dz <= 2.0; dz += 0.4) {
                const spike = new THREE.Mesh(new THREE.ConeGeometry(0.2, 0.7, 4), darkChitinMat);
                spike.rotation.x = Math.PI / 4;
                spike.position.set(0, 2.2, dz);
                spider.add(spike);
            }

            const thorax = new THREE.Mesh(new THREE.SphereGeometry(0.9, 10, 10), darkChitinMat);
            thorax.position.set(0, 1.1, -0.4);
            thorax.scale.set(1, 0.8, 1.2);
            spider.add(thorax);

            for (let i = -3; i <= 3; i += 2) {
                let eye = new THREE.Mesh(new THREE.SphereGeometry(0.08, 6, 6), glowEyeMat);
                eye.position.set(i * 0.12, 1.35, -1.1);

            fangL.rotation.x = -Math.PI / 3;
            fangL.position.set(-0.25, 0.8, -1.2);
            spider.add(fangL);

            const fangR = new THREE.Mesh(new THREE.ConeGeometry(0.1, 0.6, 4), fangMat);
            fangR.rotation.x = -Math.PI / 3;
            fangR.position.set(0.25, 0.8, -1.2);
            spider.add(fangR);

            let legBonesL = [];

            for (let i = 0; i < 4; i++) {
                let zPos = -0.7 + (i * 0.45);
                
                let legGroupL = new THREE.Group();
                legGroupL.position.set(-0.7, 1.1, zPos);

                legBonesR.push(legGroupR);

            rootGroup.add(spider);

        function updateJoystick(e) {
            const rect = joystickZone.getBoundingClientRect();
            const centerX = rect.left + rect.width / 2;
            const centerY = rect.top + rect.height / 2;
            let dx = e.clientX - centerX;
            let dy = e.clientY - centerY;
            const maxRadius = 40;
            const dist = Math.hypot(dx, dy);

            if (dist > maxRadius) {
                dx = (dx / dist) * maxRadius;
                dy = (dy / dist) * maxRadius;
            }

            joystickKnob.style.left = `${35 + dx}px`;
            joystickKnob.style.top = `${35 + dy}px`;

            joystickVector.x = dx / maxRadius;
            joystickVector.y = dy / maxRadius;
        }

                if (wi.collected) continue;
                let itemPos = new THREE.Vector3(wi.x, 0, wi.z);
                if (pPos.distanceTo(itemPos) < 4.0) {
                    wi.collected = true;
                    scene.remove(wi.group);

                    if (wi.type === 'health') {
                        playerHealth = Math.min(maxPlayerHealth, playerHealth + 45);
                        document.getElementById('playerHealth').innerText = playerHealth;

                        meatInventory += 1;
                        meatPricesInventory.push(wi.price || 100);
                        saveGameData();
                        document.getElementById('meatCount').innerText = meatInventory;
                    } else if (wi.type === 'radioactive') {
                        radioactiveInventory += 1;
                        saveGameData();
                        document.getElementById('radioactiveCount').innerText = radioactiveInventory;
                    } else if (wi.type === 'clock') {
                        let emptySlot = inventorySlots.indexOf(null);
                        if (emptySlot !== -1) {
                            inventorySlots[emptySlot] = 'clock';
                            saveGameData();
                            updateInventoryUI();
                            alert("Zaman Saati envantere eklendi! Kuşanıp SAAT butonuna basarak kullanabilirsiniz.");
                        } else {
                            alert("Envanter dolu! Saati alamadınız.");
                            wi.collected = false;
                            scene.add(wi.group);
                        }
                    }
                    return;
                }
            }

            for (let i = droppedWeapons.length - 1; i >= 0; i--) {
                let dw = droppedWeapons[i];
                if (pPos.distanceTo(dw.group.position) < 4.5) {
                    let emptySlot = inventorySlots.indexOf(null);

                        droppedWeapons.splice(i, 1);

                        updateInventoryUI();

                    return;

                return;

        function drawMinimap() {
            minimapCtx.clearRect(0, 0, 110, 110);
            minimapCtx.fillStyle = 'rgba(8, 20, 15, 0.9)';
            minimapCtx.fillRect(0, 0, 110, 110);

            minimapCtx.lineWidth = 1;
            minimapCtx.beginPath();
            minimapCtx.moveTo(55, 0); minimapCtx.lineTo(55, 110);
            minimapCtx.moveTo(0, 55); minimapCtx.lineTo(110, 55);
            minimapCtx.stroke();

            let centerX = 55, centerY = 55, scaleFactor = 0.05;

            function worldToMap(wx, wz) {
                let dx = (wx - camera.position.x) * scaleFactor; 
                let dz = (wz - camera.position.z) * scaleFactor;
                let cos = Math.cos(-yaw), sin = Math.sin(-yaw);
                return { 
                    x: centerX + (dx * cos - dz * sin), 
                    y: centerY + (dx * sin + dz * cos) 
                };
            }

            let currentHours = Math.floor(gameTimeMinutes / 60);
            let isDayTime = (currentHours >= 8 && currentHours < 19);

            if (isDayTime) {
                if (castleGodzilla && !castleGodzilla.isDead) {
                    let gPos = worldToMap(castleGodzilla.root.position.x, castleGodzilla.root.position.z);
                    if (gPos.x >= 0 && gPos.x <= 110 && gPos.y >= 0 && gPos.y <= 110) {
                        minimapCtx.fillStyle = '#ff00ff';
                        minimapCtx.fillRect(gPos.x - 3, gPos.y - 3, 6, 6);
                    }
                }

                dinos.forEach(dino => {
                    if (!dino || dino.isDead) return;
                    let mPos = worldToMap(dino.root.position.x, dino.root.position.z);
                    if (mPos.x >= 0 && mPos.x <= 110 && mPos.y >= 0 && mPos.y <= 110) {
                        minimapCtx.fillStyle = dino.hasEatenCorpse ? '#00bfff' : (dino.isAggro ? '#ff3333' : '#ffaa00');
                        minimapCtx.fillRect(mPos.x - 1.5, mPos.y - 1.5, 3, 3);
                    }
                });
            } else {
                spiders.forEach(spider => {
                    if (!spider || spider.isDead) return;

                    if (mPos.x >= 0 && mPos.x <= 110 && mPos.y >= 0 && mPos.y <= 110) {
                        minimapCtx.fillStyle = '#9400d3';
                        minimapCtx.fillRect(mPos.x - 1.5, mPos.y - 1.5, 3, 3);
                    }
                });
            }

            let butcherMap = worldToMap(butcherShop.position.x, butcherShop.position.z);
            minimapCtx.fillStyle = '#ff4747'; 
            minimapCtx.beginPath(); minimapCtx.arc(butcherMap.x, butcherMap.y, 3, 0, Math.PI * 2); minimapCtx.fill();

            let weaponMap = worldToMap(weaponShop.position.x, weaponShop.position.z);

            let weaponUpgradeMap = worldToMap(weaponUpgradeShop.position.x, weaponUpgradeShop.position.z);

            if (activeCarMesh) {

        function animate() {
            requestAnimationFrame(animate);
            const now = Date.now();

            let deltaSeconds = (now - lastClockUpdate) / 1000;
            if (deltaSeconds >= 0.05) { 
                gameTimeMinutes += deltaSeconds * 3.0; 
                lastClockUpdate = now;

                if (gameTimeMinutes >= 24 * 60) {
                    gameTimeMinutes %= 24 * 60;
                    currentDay++;
                    document.getElementById('dayCountVal').innerText = currentDay;
                    updateSpiderCountForDay();
                }

                let hours = Math.floor(gameTimeMinutes / 60);
                let minutes = Math.floor(gameTimeMinutes % 60);
                let hoursStr = String(hours).padStart(2, '0');
                let minsStr = String(minutes).padStart(2, '0');
                document.getElementById('clockVal').innerText = `${hoursStr}:${minsStr}`;

                let isDayTime = (hours >= 8 && hours < 19);

                let timeNormalized = gameTimeMinutes / (24 * 60); 
                let sunAngle = timeNormalized * Math.PI * 2;
                let sunX = Math.cos(sunAngle) * 300;
                let sunY = Math.sin(sunAngle) * 300;
                sun.position.set(sunX, Math.max(-50, sunY), 200);
                sunMesh.position.set(sunX * 1.5, sunY * 1.5, 300);

                if (isDayTime) {
                    let intensity = Math.min(1.2, Math.max(0.1, sunY / 150));
                    sun.intensity = intensity;
                    ambientLight.intensity = Math.max(0.2, intensity * 0.6);
                    scene.background.setHex(0x4a739b);
                    scene.fog.color.setHex(0x4a739b);

                    spiders.forEach(s => { if (s && s.root) s.root.visible = false; });
                    dinos.forEach(d => { if (d && d.root) d.root.visible = true; });
                    if (castleGodzilla && castleGodzilla.root) castleGodzilla.root.visible = true;
                } else {
                    sun.intensity = 0.05;
                    ambientLight.intensity = 0.1;
                    scene.background.setHex(0x050b14);
                    scene.fog.color.setHex(0x050b14);

                    spiders.forEach(s => { if (s && s.root) s.root.visible = true; });
                    dinos.forEach(d => { if (d && d.root) d.root.visible = false; });
                    if (castleGodzilla && castleGodzilla.root) castleGodzilla.root.visible = false;
                }
            }

            if (respawnProtectionTimer > 0) {
                respawnProtectionTimer -= deltaSeconds;
            }

            for (let i = bloodParticles.length - 1; i >= 0; i--) {
                let p = bloodParticles[i]; p.mesh.position.add(p.vel); p.vel.y -= 0.012; p.life -= 0.025;
                if (p.mesh.position.y <= 0.05 || p.life <= 0) { scene.remove(p.mesh); bloodParticles.splice(i, 1); }
            }

            if (playerHealth < maxPlayerHealth && now - lastDamageTime > 5000) {
                document.getElementById('regenStatus').innerText = "(Can Yenileniyor...)";
                if (now - lastRegenTick > 1000) { 
                    playerHealth = Math.min(maxPlayerHealth, playerHealth + 10); 
                    document.getElementById('playerHealth').innerText = playerHealth; 
                    lastRegenTick = now; 
                }
            } else if (playerHealth < maxPlayerHealth) {
                document.getElementById('regenStatus').innerText = `(${Math.ceil((5000 - (now - lastDamageTime)) / 1000)}s sonra regen)`;
            } else document.getElementById('regenStatus').innerText = "";

            document.getElementById('cooldownContainer').style.display = 'block';

            let currentHours = Math.floor(gameTimeMinutes / 60);
            let isDayTime = (currentHours >= 8 && currentHours < 19);

            if (isDriving && activeCarMesh) {
                let maxCarSpeed = activeCarData.baseSpeed * (1 + (activeCarData.speedLevel / 20) * 4);
                
                let moveY = -joystickVector.y;
                let moveX = joystickVector.x;

                if (moveY > 0.1) carVelocity = Math.min(maxCarSpeed, carVelocity + 0.015 * moveY);
                else if (moveY < -0.1) carVelocity = Math.max(-maxCarSpeed * 0.4, carVelocity + 0.015 * moveY);
                else carVelocity *= 0.95;

                if (Math.abs(moveX) > 0.1) activeCarMesh.rotation.y -= 0.035 * moveX * (carVelocity !== 0 ? Math.sign(carVelocity) : 0);

                activeCarMesh.translateZ(-carVelocity);

                let carTopBackOffset = new THREE.Vector3(0, 5.5, -8).applyAxisAngle(new THREE.Vector3(0, 1, 0), activeCarMesh.rotation.y + carCameraYawOffset);
                let targetCameraPos = activeCarMesh.position.clone().add(carTopBackOffset);
                targetCameraPos.y += carCameraPitchOffset * 3;
                camera.position.copy(targetCameraPos);
                
                let lookTarget = activeCarMesh.position.clone().add(new THREE.Vector3(0, 1.2, 0));
                camera.lookAt(lookTarget);

                document.getElementById('interactionText').style.display = 'block';
                document.getElementById('interactionText').innerText = "Araçtan İnmek İçin ARABA İN Butonuna Basın";
            } else if (isPlaying && playerHealth > 0) {
                let isCrouching = keys['Crouch'];
                let isRunning = keys['Run'];
                
                let baseSpeed = isCrouching ? 0.11 : (isRunning ? 0.42 : 0.22);
                let speedMultiplier = 1 + (playerSpeedLevel / 20) * 4;
                let effectiveSpeed = baseSpeed * speedMultiplier;

                let targetHeight = isCrouching ? 1.2 : 2.0;
                camera.position.y = THREE.MathUtils.lerp(camera.position.y, targetHeight, 0.2);

                let oldX = camera.position.x, oldZ = camera.position.z;

                let moveX = joystickVector.x;
                let moveZ = joystickVector.y;

                if (Math.abs(moveZ) > 0.05) camera.translateZ(moveZ * effectiveSpeed);
                if (Math.abs(moveX) > 0.05) camera.translateX(moveX * effectiveSpeed);

                let localX = camera.position.x - 0;
                let localZ = camera.position.z - 35;
                let insideHousePlatform = (localX >= -4.5 && localX <= 4.5 && localZ >= -4.5 && localZ <= 4.5);
                
                if (insideHousePlatform) {
                    camera.position.y = Math.max(camera.position.y, 10.3);
                } else {
                    let onLadder = (localX >= -1.2 && localX <= 1.2 && localZ >= 3.8 && localZ <= 5.2);
                    if (!onLadder && camera.position.y > 2.5 && !insideHousePlatform) {
                        camera.position.x = oldX; camera.position.z = oldZ;
                    }
                }

                for (let box of castleColliders) {
                    if (camera.position.x >= box.minX && camera.position.x <= box.maxX && camera.position.z >= box.minZ && camera.position.z <= box.maxZ) {
                        camera.position.x = oldX; camera.position.z = oldZ; break;
                    }
                }

                for (let tree of treeColliders) {
                    if (Math.abs(camera.position.x - tree.x) < 3.5 && Math.abs(camera.position.z - tree.z) < 3.5) {
                        let dx = camera.position.x - tree.x, dz = camera.position.z - tree.z;
                        if (Math.sqrt(dx * dx + dz * dz) < tree.radius) { camera.position.x = oldX; camera.position.z = oldZ; break; }
                    }
                }

                for (let bldg of buildingColliders) {
                    let halfW = bldg.width / 2, halfD = bldg.depth / 2;
                    if (camera.position.x > bldg.x - halfW && camera.position.x < bldg.x + halfW && camera.position.z > bldg.z - halfD && camera.position.z < bldg.z + halfD) {
                        camera.position.x = oldX; camera.position.z = oldZ; break;
                    }
                }

                camera.position.y += playerVelocityY; playerVelocityY += gravity;
                
                let floorLimit = isCrouching ? 1.2 : 2.0;
                if (insideHousePlatform && camera.position.y >= 9.8) {
                    floorLimit = 10.3;
                }

                if (camera.position.y <= floorLimit) { camera.position.y = floorLimit; playerVelocityY = 0; isGrounded = true; }

                let showInteract = false; const pPos = camera.position;
                let nearCorpse = false;
                let targetCorpseDino = null;

                if (isDayTime && castleGodzilla && castleGodzilla.isDead && !castleGodzilla.hasBeenSkin) {
                    if (pPos.distanceTo(castleGodzilla.root.position) < 10.0) {
                        showInteract = true;
                        nearCorpse = true;
                        targetCorpseDino = castleGodzilla;
                    }
                }

                if (isDayTime) {
                    dinos.forEach(dino => {
                        if (dino && dino.isDead && !dino.hasBeenSkin) {
                            if (pPos.distanceTo(dino.root.position) < 5.0) {
                                showInteract = true;
                                nearCorpse = true;
                                targetCorpseDino = dino;
                            }
                        }
                    });
                }

                if (nearCorpse && isHoldingG && targetCorpseDino) {
                    if (currentSkinningDino !== targetCorpseDino) {
                        currentSkinningDino = targetCorpseDino;
                        skinningProgress = 0;
                    }
                    skinningProgress += deltaSeconds;
                    let pBar = document.getElementById('skinningProgressBar');
                    let pFill = document.getElementById('skinningProgressFill');
                    pBar.style.display = 'block';
                    let percent = Math.min(100, (skinningProgress / targetCorpseDino.skinningDuration) * 100);
                    pFill.style.width = percent + '%';

                    if (skinningProgress >= targetCorpseDino.skinningDuration) {
                        targetCorpseDino.hasBeenSkin = true;
                        leatherInventory += targetCorpseDino.leatherAmount;
                        document.getElementById('leatherCount').innerText = leatherInventory;
                        
                        if (targetCorpseDino.root) scene.remove(targetCorpseDino.root);
                        if (targetCorpseDino.puddleMesh) scene.remove(targetCorpseDino.puddleMesh);

                        pBar.style.display = 'none';
                        currentSkinningDino = null;
                        skinningProgress = 0;
                    }
                } else {
                    document.getElementById('skinningProgressBar').style.display = 'none';
                    if (!isHoldingG) { currentSkinningDino = null; skinningProgress = 0; }
                }

                if (pPos.distanceTo(butcherShop.position) < 9 || pPos.distanceTo(weaponShop.position) < 9 || pPos.distanceTo(weaponUpgradeShop.position) < 9 || pPos.distanceTo(weaponDamageUpgradeShop.position) < 9 || pPos.distanceTo(leatherShop.position) < 9 || pPos.distanceTo(carShop.position) < 9 || pPos.distanceTo(upgradeShop.position) < 9) showInteract = true;
                if (pPos.distanceTo(new THREE.Vector3(-2.8, 10.5, 32.5)) < 4.5) showInteract = true;
                if (activeCarMesh && pPos.distanceTo(activeCarMesh.position) < 5.0) showInteract = true;
                
                let nearItem = false;
                worldItems.forEach(wi => {
                    if (!wi.collected && pPos.distanceTo(new THREE.Vector3(wi.x, 0, wi.z)) < 4.0) {
                        showInteract = true;
                        nearItem = true;
                    }
                });

                let nearWeapon = false;
                droppedWeapons.forEach(dw => {
                    if (pPos.distanceTo(dw.group.position) < 4.5) {
                        showInteract = true;
                        nearWeapon = true;
                    }
                });

                droppedMeats.forEach(dm => { if (pPos.distanceTo(dm.group.position) < 4.5) showInteract = true; });
                droppedRadioactive.forEach(dr => { if (pPos.distanceTo(dr.group.position) < 4.5) showInteract = true; });
                
                let interactText = "E / G Butonuna basarak etkileşime geç";
                if (nearCorpse) interactText = "Leşi deriye çevirmek için E / G Butonuna BASILI TUTUN!";
                if (nearItem) interactText = "Eşyayı almak için E / G Butonuna basın";
                if (nearWeapon) interactText = "Silahı almak için E / G Butonuna basın";
                if (pPos.distanceTo(new THREE.Vector3(-2.8, 10.5, 32.5)) < 4.5) interactText = "Uyumak / Sabahı Beklemek için E / G Butonuna basın";
                if (activeCarMesh && pPos.distanceTo(activeCarMesh.position) < 5.0) interactText = "Araca binmek için E / G Butonuna basın";
                document.getElementById('interactionText').style.display = showInteract ? 'block' : 'none';
                document.getElementById('interactionText').innerText = interactText;
                btnInteract.style.display = showInteract ? 'flex' : 'none';

                if (isDayTime) {
                    if (castleGodzilla && !castleGodzilla.isDead) {
                        if (castleGodzilla.isAggro) {
                            castleGodzilla.root.lookAt(camera.position.x, 0, camera.position.z);
                            castleGodzilla.root.translateZ(castleGodzilla.speed);
                            castleGodzilla.root.position.x = Math.max(-140, Math.min(140, castleGodzilla.root.position.x));
                            castleGodzilla.root.position.z = Math.max(castleZ - 90, Math.min(castleZ + 90, castleGodzilla.root.position.z));

                            if (castleGodzilla.root.position.distanceTo(camera.position) < 10.0 && now - lastAttackTime > 1000) {
                                let dmgTaken = castleGodzilla.damage;
                                if (playerArmor > 0) {
                                    dmgTaken = Math.round(dmgTaken * 0.5);
                                    playerArmor = Math.max(0, playerArmor - castleGodzilla.damage);
                                    updateArmorUI();
                                }
                                playerHealth -= dmgTaken;
                                document.getElementById('playerHealth').innerText = playerHealth;
                                lastAttackTime = now; lastDamageTime = now;

                                const overlay = document.getElementById('damageOverlay');
                                overlay.style.opacity = '1';
                                setTimeout(() => { overlay.style.opacity = '0'; }, 120);

                                if (playerHealth <= 0) {
                                    alert("ÖLDÜNÜZ! Godzilla sizi ezdi.");
                                    playerHealth = maxPlayerHealth;
                                    document.getElementById('playerHealth').innerText = playerHealth;
                                    camera.position.set(0, 2, 25);
                                    respawnProtectionTimer = 10.0;
                                }
                            }
                        }
                    }

                    let allCorpses = [];
                    if (castleGodzilla && castleGodzilla.isDead && !castleGodzilla.hasBeenSkin) {
                        allCorpses.push(castleGodzilla);
                    }
                    dinos.forEach(d => {
                        if (d && d.isDead && !d.hasBeenSkin) allCorpses.push(d);
                    });

                    dinos.forEach((dino) => {
                        if (!dino) return;

                        if (dino.isDead) {
                            if (dino.deathProgress < 1.0) {
                                dino.deathProgress += deltaSeconds * 1.5;
                                dino.root.position.y = Math.max(0, dino.root.position.y - deltaSeconds * 3.0);
                                dino.mesh.rotation.z = THREE.MathUtils.lerp(dino.mesh.rotation.z, Math.PI / 2, 0.1);
                                dino.mesh.rotation.x = THREE.MathUtils.lerp(dino.mesh.rotation.x, 0.2, 0.1);
                            }
                            return;
                        }

                        dino.animTime += dino.isAggro ? 0.35 : 0.18;

                        if (dino.legL && dino.legR) {
                            let legSwing = Math.sin(dino.animTime * 6.0) * 0.6;
                            dino.legL.rotation.x = legSwing;
                            dino.legR.rotation.x = -legSwing;
                        }
                        dino.mesh.position.y = Math.abs(Math.sin(dino.animTime * 6.0)) * 0.25;

                        if (dino.isFlying) {
                            const flapAngle = Math.sin(dino.animTime * 3.5) * 1.1;
                            if (dino.wingL && dino.wingR) {
                                dino.wingL.rotation.z = flapAngle;
                                dino.wingR.rotation.z = -flapAngle;
                            }
                            dino.targetPos.y = 8 + Math.sin(dino.animTime * 0.6) * 10;
                        }

                        if (dino.isAggro) {
                            dino.root.lookAt(camera.position.x, dino.isFlying ? dino.root.position.y : 0, camera.position.z);
                            dino.root.translateZ(dino.speed);

                            let attackDist = dino.isFlying ? 5.5 : (3.8 * dino.scale);
                            if (dino.root.position.distanceTo(camera.position) < attackDist && now - lastAttackTime > 800) {
                                let dmgTaken = dino.damage;
                                if (playerArmor > 0) {
                                    dmgTaken = Math.round(dmgTaken * 0.5);
                                    playerArmor = Math.max(0, playerArmor - dino.damage);
                                    updateArmorUI();
                                }
                                playerHealth -= dmgTaken; 
                                document.getElementById('playerHealth').innerText = playerHealth;
                                lastAttackTime = now; lastDamageTime = now;

                                const overlay = document.getElementById('damageOverlay');
                                overlay.style.opacity = '1';
                                setTimeout(() => { overlay.style.opacity = '0'; }, 120);

                                if (playerHealth <= 0) { 
                                    alert("ÖLDÜNÜZ! Yeniden doğdunuz."); 
                                    playerHealth = maxPlayerHealth;
                                    document.getElementById('playerHealth').innerText = playerHealth;
                                    camera.position.set(0, 2, 25);
                                    respawnProtectionTimer = 10.0; 
                                }
                            }
                        } else {
                            let targetCorpse = null;
                            let minCorpseDist = 400;

                            if (!dino.hasEatenCorpse && allCorpses.length > 0) {
                                for (let corpse of allCorpses) {
                                    if (corpse.eatersCount < 2) {
                                        let dist = dino.root.position.distanceTo(corpse.root.position);
                                        if (dist < minCorpseDist) {
                                            minCorpseDist = dist;
                                            targetCorpse = corpse;
                                        }
                                    }
                                }
                            }

                            if (targetCorpse) {
                                dino.root.lookAt(targetCorpse.root.position.x, 0, targetCorpse.root.position.z);
                                dino.root.translateZ(dino.speed * 0.8);

                                if (dino.root.position.distanceTo(targetCorpse.root.position) < 6.0) {
                                    if (!dino.hasEatenCorpse) {
                                        dino.hasEatenCorpse = true;
                                        targetCorpse.eatersCount++;
                                        
                                        dino.maxHealth = Math.round(dino.maxHealth * 1.5);
                                        dino.health = dino.maxHealth;
                                        dino.speed *= 1.4;

                                        if (dino.uiBar) {
                                            dino.uiBar.style.backgroundColor = '#00bfff';
                                        }
                                        if (dino.uiText) {
                                            dino.uiText.innerText = `${Math.round(dino.health)} / ${dino.maxHealth}`;
                                        }

                                        targetCorpse.hasBeenSkin = true;
                                        if (targetCorpse.root) scene.remove(targetCorpse.root);
                                        if (targetCorpse.puddleMesh) scene.remove(targetCorpse.puddleMesh);
                                        if (targetCorpse.uiElement) targetCorpse.uiElement.style.display = 'none';
                                    }
                                }
                            } else {
                                let distToTarget = dino.root.position.distanceTo(dino.targetPos);
                                dino.changeDirTimer++;

                                if (distToTarget < 4 || dino.changeDirTimer > 200) {
                                    let newX = Math.max(-1150, Math.min(1150, dino.root.position.x + (Math.random() - 0.5) * 180));
                                    let newZ = Math.max(-1150, Math.min(1150, dino.root.position.z + (Math.random() - 0.5) * 180));
                                    let newY = dino.isFlying ? (8 + Math.random() * 15) : 0;
                                    dino.targetPos.set(newX, newY, newZ);
                                    dino.changeDirTimer = 0;
                                }

                                let currentRot = dino.root.rotation.y;
                                dino.root.lookAt(dino.targetPos.x, dino.isFlying ? dino.targetPos.y : 0, dino.targetPos.z);
                                let targetRot = dino.root.rotation.y;
                                dino.root.rotation.y = THREE.MathUtils.lerp(currentRot, targetRot, 0.08);

                                dino.root.translateZ(dino.speed * 0.55);
                            }
                        }
                    });
                } else {
                    spiders.forEach(spider => {
                        if (!spider || spider.isDead) return;
                        spider.animTime += 0.3;
                        let legWave = Math.sin(spider.animTime * 8.0) * 0.4;
                        if (spider.legBonesL && spider.legBonesR) {
                            spider.legBonesL.forEach((b, idx) => { b.rotation.z = (idx % 2 === 0 ? 1 : -1) * legWave; });
                            spider.legBonesR.forEach((b, idx) => { b.rotation.z = (idx % 2 === 0 ? -1 : 1) * legWave; });
                        }

                        spider.root.lookAt(camera.position.x, 0, camera.position.z);
                        spider.root.translateZ(spider.speed);

                        if (spider.root.position.distanceTo(camera.position) < 4.5 * spider.scale && now - lastAttackTime > 800) {
                            let dmgTaken = spider.damage;
                            if (playerArmor > 0) {
                                dmgTaken = Math.round(dmgTaken * 0.5);
                                playerArmor = Math.max(0, playerArmor - spider.damage);
                                updateArmorUI();
                            }
                            playerHealth -= dmgTaken;
                            document.getElementById('playerHealth').innerText = playerHealth;
                            lastAttackTime = now; lastDamageTime = now;

                            const overlay = document.getElementById('damageOverlay');
                            overlay.style.opacity = '1';
                            setTimeout(() => { overlay.style.opacity = '0'; }, 120);

                            if (playerHealth <= 0) {
                                alert("ÖLDÜNÜZ! Dev korkunç örümcekler sizi avladı.");
                                playerHealth = maxPlayerHealth;
                                document.getElementById('playerHealth').innerText = playerHealth;
                                camera.position.set(0, 2, 25);
                                respawnProtectionTimer = 10.0;
                            }
                        }
                    });
                }
            }

            updateDinoHealthBars();
            drawMinimap();
            renderer.render(scene, camera);
        }

        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });

        animate();
    </script>
